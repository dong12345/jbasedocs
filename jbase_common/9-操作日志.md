# 操作日志

JBase.Common 提供特性驱动的操作日志记录机制，自动记录用户的关键操作行为。

## 核心组件

| 组件 | 说明 |
|------|------|
| `OperationLogAttribute` | 操作日志特性 |
| `OperationLogActionFilter` | 操作日志过滤器 |
| `OperationLogEntry` | 日志条目 |
| `OperationLogOptions` | 配置选项 |

## OperationLogAttribute

### 基本用法

```csharp
[ApiController]
[Route("api/[controller]")]
public class UserController : ControllerBase
{
    [HttpPost]
    [OperationLog(ModuleName = "用户管理", OperationType = OperationTypeEnum.Insert)]
    public async Task<IActionResult> Create([FromBody] CreateUserDto dto)
    {
        return Ok();
    }

    [HttpPut("{id}")]
    [OperationLog(ModuleName = "用户管理", OperationType = OperationTypeEnum.Update)]
    public async Task<IActionResult> Update(int id, [FromBody] UpdateUserDto dto)
    {
        return Ok();
    }

    [HttpDelete("{id}")]
    [OperationLog(ModuleName = "用户管理", OperationType = OperationTypeEnum.Delete)]
    public async Task<IActionResult> Delete(int id)
    {
        return Ok();
    }

    [HttpGet("{id}")]
    [OperationLog(ModuleName = "用户管理", OperationType = OperationTypeEnum.Detail)]
    public async Task<IActionResult> GetDetail(int id)
    {
        return Ok();
    }

    [HttpGet]
    [OperationLog(ModuleName = "用户管理", OperationType = OperationTypeEnum.List)]
    public async Task<IActionResult> GetList()
    {
        return Ok();
    }
}
```

### 完整配置

```csharp
[OperationLog(
    ModuleName = "用户管理",
    OperationType = OperationTypeEnum.Insert,
    CustomOperationName = "新增用户",           // 自定义操作名称（优先级高于 OperationType）
    FilterSensitiveFields = new[] { "Password", "SecretKey" }  // 过滤敏感字段
)]
public async Task<IActionResult> Create([FromBody] CreateUserDto dto)
{
    // ...
}
```

## OperationTypeEnum 枚举

```csharp
public enum OperationTypeEnum
{
    Other,    // 其它
    Insert,   // 新增
    Update,   // 修改
    Delete,   // 删除
    Detail,   // 详情
    List,     // 列表
}
```

> **注意**：如果预定义的枚举值不满足需求，可以使用 `CustomOperationName` 自定义操作名称。

## 配置选项

```csharp
services.Configure<OperationLogOptions>(options =>
{
    // 是否启用文本日志记录（默认false）
    options.EnableTxtLogging = false;

    // 是否启用数据库日志记录（默认false）
    options.EnableDatabaseLogging = false;

    // 是否对Url中的参数值解码
    options.IsUrlDecode = false;

    // 是否记录请求头信息（默认false）
    options.RecordRequestHeaders = false;

    // 是否记录响应头信息（默认false）
    options.RecordResponseHeaders = false;

    // 需要过滤的敏感头信息列表
    options.FilterSensitiveHeaders = new[] { "Authorization", "Cookie" };

    // 自定义数据提供器（用于填充ExtraData字段）
    options.ExtraDataProvider = context => "";
});
```

## OperationLogEntry 日志条目

记录时会生成 `OperationLogEntry` 对象：

```csharp
public class OperationLogEntry
{
    /// <summary>
    /// 操作用户ID
    /// </summary>
    public string UserId { get; set; }

    /// <summary>
    /// 操作用户名
    /// </summary>
    public string UserName { get; set; }

    /// <summary>
    /// 模块名称
    /// </summary>
    public string ModuleName { get; set; }

    /// <summary>
    /// 操作类型
    /// </summary>
    public OperationTypeEnum OperationType { get; set; }

    /// <summary>
    /// 自定义操作名称
    /// </summary>
    public string CustomOperationName { get; set; }

    /// <summary>
    /// 请求方法
    /// </summary>
    public string Method { get; set; }

    /// <summary>
    /// 请求路径
    /// </summary>
    public string Path { get; set; }

    /// <summary>
    /// 输入参数（已脱敏）
    /// </summary>
    public string InputParameters { get; set; }

    /// <summary>
    /// 输出结果
    /// </summary>
    public string OutputResult { get; set; }

    /// <summary>
    /// IP 地址
    /// </summary>
    public string IP { get; set; }

    /// <summary>
    /// 用户代理
    /// </summary>
    public string UserAgent { get; set; }

    /// <summary>
    /// TraceId
    /// </summary>
    public string RequestId { get; set; }

    /// <summary>
    /// 执行时间（毫秒）
    /// </summary>
    public long ElapsedMilliseconds { get; set; }

    /// <summary>
    /// 操作状态
    /// </summary>
    public bool Success { get; set; }

    /// <summary>
    /// 错误消息
    /// </summary>
    public string ErrorMsg { get; set; }

    /// <summary>
    /// 操作时间
    /// </summary>
    public DateTime OperationTime { get; set; }
}
```

## 实现日志写入

需要实现 `IOperationLogDatabaseWriter` 接口：

```csharp
public class SqlOperationLogWriter : IOperationLogDatabaseWriter
{
    private readonly AppDbContext _dbContext;
    private readonly ILogger<SqlOperationLogWriter> _logger;

    public SqlOperationLogWriter(AppDbContext dbContext, ILogger<SqlOperationLogWriter> logger)
    {
        _dbContext = dbContext;
        _logger = logger;
    }

    public async Task WriteLogAsync(OperationLogEntry entry)
    {
        try
        {
            var log = new OperationLog
            {
                UserId = entry.UserId,
                UserName = entry.UserName,
                ModuleName = entry.ModuleName,
                OperationType = entry.OperationType.ToString(),
                OperationName = entry.CustomOperationName,
                Method = entry.Method,
                Path = entry.Path,
                InputParameters = entry.InputParameters,
                OutputResult = entry.OutputResult,
                IP = entry.IP,
                UserAgent = entry.UserAgent,
                RequestId = entry.RequestId,
                ElapsedMilliseconds = entry.ElapsedMilliseconds,
                Success = entry.Success,
                ErrorMsg = entry.ErrorMsg,
                OperationTime = entry.OperationTime
            };

            _dbContext.OperationLogs.Add(log);
            await _dbContext.SaveChangesAsync();
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "写入操作日志失败");
        }
    }
}
```

### 注册实现

```csharp
services.AddScoped<IOperationLogDatabaseWriter, SqlOperationLogWriter>();
```

## 注册过滤器

### 方式一：全局注册

```csharp
public void ConfigureServices(IServiceCollection services)
{
    services.AddControllers(options =>
    {
        options.Filters.Add<OperationLogActionFilter>();
    });
}
```

### 方式二：按需注册

```csharp
public void ConfigureServices(IServiceCollection services)
{
    services.AddScoped<OperationLogActionFilter>();
}
```

## 完整使用示例

### 1. 定义 DTO

```csharp
public class CreateOrderDto
{
    public string CustomerName { get; set; }
    public decimal Amount { get; set; }
    public string Remark { get; set; }
}

public class OrderDto
{
    public int Id { get; set; }
    public string OrderNo { get; set; }
    public string CustomerName { get; set; }
    public decimal Amount { get; set; }
    public string Status { get; set; }
}
```

### 2. 创建控制器

```csharp
[ApiController]
[Route("api/[controller]")]
public class OrderController : ControllerBase
{
    private readonly IOrderService _orderService;

    public OrderController(IOrderService orderService)
    {
        _orderService = orderService;
    }

    [HttpPost]
    [OperationLog(
        ModuleName = "订单管理",
        OperationType = OperationTypeEnum.Insert,
        CustomOperationName = "创建订单"
    )]
    public async Task<IActionResult> Create([FromBody] CreateOrderDto dto)
    {
        var result = await _orderService.CreateAsync(dto);
        return Ok(result);
    }

    [HttpPut("{id}")]
    [OperationLog(
        ModuleName = "订单管理",
        OperationType = OperationTypeEnum.Update,
        CustomOperationName = "修改订单"
    )]
    public async Task<IActionResult> Update(int id, [FromBody] CreateOrderDto dto)
    {
        await _orderService.UpdateAsync(id, dto);
        return Ok();
    }

    [HttpDelete("{id}")]
    [OperationLog(
        ModuleName = "订单管理",
        OperationType = OperationTypeEnum.Delete,
        CustomOperationName = "删除订单"
    )]
    public async Task<IActionResult> Delete(int id)
    {
        await _orderService.DeleteAsync(id);
        return Ok();
    }

    [HttpPost("{id}/cancel")]
    [OperationLog(
        ModuleName = "订单管理",
        OperationType = OperationTypeEnum.Update,
        CustomOperationName = "取消订单"
    )]
    public async Task<IActionResult> Cancel(int id)
    {
        await _orderService.CancelAsync(id);
        return Ok();
    }

    [HttpGet("export")]
    [OperationLog(
        ModuleName = "订单管理",
        OperationType = OperationTypeEnum.Other,
        CustomOperationName = "导出订单"
    )]
    public async Task<IActionResult> Export([FromQuery] DateTime startDate, [FromQuery] DateTime endDate)
    {
        var file = await _orderService.ExportAsync(startDate, endDate);
        return File(file, "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet");
    }
}
```

### 3. 查看日志记录

```sql
SELECT
    OperationTime,
    UserName,
    ModuleName,
    OperationName,
    Method,
    Path,
    IpAddress,
    ElapsedMilliseconds,
    Status
FROM OperationLogs
WHERE ModuleName = '订单管理'
ORDER BY OperationTime DESC;
```

输出：

## 最佳实践

1. **统一模块命名**：保持模块名称一致性
2. **详细操作描述**：使用 `CustomOperationName` 提供更友好的描述
3. **敏感字段过滤**：敏感信息使用 `FilterSensitiveFields`
4. **异步写入**：日志写入应异步执行，不阻塞主流程
5. **批量写入**：高并发场景可考虑批量写入
6. **日志保留策略**：根据业务需求设置保留周期
