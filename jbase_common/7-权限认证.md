# 权限认证

JBase.Common 提供基于操作码的细粒度权限验证机制。

## 核心组件

| 组件 | 说明 |
|------|------|
| `AppAuthorizeAttribute` | 操作权限特性 |
| `AppRoleAuthorizeAttribute` | 角色权限特性 |
| `AppAuthorizationMiddlewareBase` | 授权中间件基类 |

## AppAuthorizeAttribute

### 基本用法

```csharp
[ApiController]
[Route("api/[controller]")]
public class UserController : ControllerBase
{
    // 需要 User.Create 权限
    [HttpPost]
    [AppAuthorize("User.Create")]
    public async Task<IActionResult> Create([FromBody] CreateUserDto dto)
    {
        return Ok();
    }

    // 需要 User.Update 权限
    [HttpPut("{id}")]
    [AppAuthorize("User.Update")]
    public async Task<IActionResult> Update(int id, [FromBody] UpdateUserDto dto)
    {
        return Ok();
    }

    // 需要 User.Delete 权限
    [HttpDelete("{id}")]
    [AppAuthorize("User.Delete")]
    public async Task<IActionResult> Delete(int id)
    {
        return Ok();
    }

    // 需要 User.View 权限
    [HttpGet]
    [AppAuthorize("User.View")]
    public async Task<IActionResult> GetList()
    {
        return Ok();
    }
}
```

### 类级别授权

```csharp
// 整个控制器需要授权
[ApiController]
[Route("api/[controller]")]
[AppAuthorize("Admin")]  // 需要 Admin 权限
public class SystemController : ControllerBase
{
    // 所有方法都需要 Admin 权限

    [HttpGet("users")]
    public IActionResult GetUsers() => Ok();

    [HttpGet("settings")]
    public IActionResult GetSettings() => Ok();
}
```

### 多权限需求

```csharp
// 同时需要多个权限（AND 关系）
[HttpPost("export")]
[AppAuthorize("Order.View")]
[AppAuthorize("Order.Export")]
public async Task<IActionResult> Export()
{
    return Ok();
}
```

### 自定义错误消息

```csharp
[HttpDelete("{id}")]

    return Ok();

```

## AppRoleAuthorizeAttribute

### 基本用法

```csharp
[ApiController]
[Route("api/[controller]")]
public class AdminController : ControllerBase
{
    // 仅限管理员角色
    [HttpGet("stats")]
    [AppRoleAuthorize("Admin")]
    public IActionResult GetStats() => Ok();

    // 支持多个角色（满足其一即可）
    [HttpGet("manage")]
    [AppRoleAuthorize("Admin", "SuperAdmin")]
    public IActionResult Manage() => Ok();
}
```

## 实现权限验证逻辑

通过继承 `AppAuthorizationMiddlewareBase` 中间件基类，实现具体的权限验证逻辑。

### 自定义授权中间件

```csharp
using System.Net;
using JBase.Common.Authorization;
using JBase.Common.Extensions;
using JBase.Common.Middleware;
using JBaseWeb.FreeSql.Test.Common;

namespace JBaseWeb.FreeSql.Test
{
    /// <summary>
    /// 统一授权中间件（集中式：登录状态 + 角色 + 操作权限）
    /// </summary>
    public class CustomAuthorizeMiddleware : AppAuthorizationMiddlewareBase
    {
        /// <summary>
        /// 所有权限标识，表示拥有系统内所有操作权限
        /// </summary>
        private const string AllPermission = "*:*:*";

        public CustomAuthorizeMiddleware(RequestDelegate next) : base(next)
        {
        }

        /// <summary>
        /// 处理授权校验的核心方法
        /// </summary>
        protected override Task<AppAuthorizeResult> HandleAuthorizeAsync(HttpContext context)
        {
            // 1. 获取当前终结点上的角色和操作权限特性
            var roleAuthorize = context.GetMetadata<AppRoleAuthorizeAttribute>();
            var permissionAuthorize = context.GetMetadata<AppAuthorizeAttribute>();

            // 如果既没有角色特性也没有操作权限特性，则直接通过
            if (roleAuthorize == null && permissionAuthorize == null)
            {
                return Task.FromResult(AppAuthorizeResult.Success());
            }

            // 2. 从缓存中获取当前登录用户信息
            var loginUser = SecurityUtils.GetLoginUser();

            // 3. 角色授权检查
            if (roleAuthorize != null && roleAuthorize.Roles?.Length > 0)
            {
                var userRoles = loginUser.Roles ?? new List<string>();
                var hasRequiredRole = roleAuthorize.Roles
                    .Any(r => userRoles.Contains(r, StringComparer.OrdinalIgnoreCase));

                if (!hasRequiredRole)
                {
                    return Task.FromResult(AppAuthorizeResult.Failed("无访问权限", (int)HttpStatusCode.Forbidden));
                }
            }

            // 4. 操作级权限检查
            if (permissionAuthorize != null)
            {
                var hasPermission = HasOperationPermission(loginUser, permissionAuthorize.Code);
                if (!hasPermission)
                {
                    var errorMessage = string.IsNullOrWhiteSpace(permissionAuthorize.Message)
                        ? "无操作权限"
                        : permissionAuthorize.Message;
                    return Task.FromResult(AppAuthorizeResult.Failed(errorMessage, (int)HttpStatusCode.Forbidden));
                }
            }

            // 5. 全部校验通过
            return Task.FromResult(AppAuthorizeResult.Success());
        }

        /// <summary>
        /// 判断当前登录用户是否拥有指定操作权限
        /// </summary>
        private bool HasOperationPermission(LoginUser loginUser, string permissionCode)
        {
            if (string.IsNullOrWhiteSpace(permissionCode))
                return false;

            var permissions = loginUser.Permissions ?? new List<string>();

            // 拥有所有权限标识或包含指定权限编码则视为有权限
            return permissions.Contains(AllPermission, StringComparer.OrdinalIgnoreCase) ||
                   permissions.Contains(permissionCode.Trim(), StringComparer.OrdinalIgnoreCase);
        }
    }
}
```

### 注册中间件

```csharp
public void Configure(IApplicationBuilder app)
{
    // 基础中间件
    app.UseTraceId();
    app.UseCors();
    app.UseExceptionMiddleware();

    // Swagger（开发环境）
    if (app.Environment.IsDevelopment())
    {
        app.UseSwagger();
        app.UseSwaggerUI();
    }

    app.UseRouting();

    // API鉴权（仅对 /api 路径生效）
    app.UseWhen(ctx => ctx.Request.Path.StartsWithSegments("/api"), apiApp =>
    {
        apiApp.UseJwt();
        apiApp.UseMiddleware<CustomAuthorizeMiddleware>();
    });

    app.MapControllers();
}
```

> **注意**：使用 `UseWhen` 可以让授权中间件仅对 `/api` 路径生效，非 API 路径（如 SignalR、静态文件等）不会被拦截。

### AppAuthorizationMiddlewareBase 基类说明

| 方法 | 说明 |
|------|------|
| `ShouldSkipAsync` | 判断是否跳过授权校验（已处理 Swagger、AllowAnonymous） |
| `HandleAuthorizeAsync` | 核心授权校验方法，**子类重写此方法** |
| `GetEndpoint` | 获取当前终结点 |
| `GetMetadata<T>` | 从终结点获取指定的元数据 |
| `WriteForbiddenResponse` | 写入拒绝访问的响应 |

## 权限存储设计

### 数据库表设计建议

```sql
-- 角色表
CREATE TABLE Role (
    Id INT PRIMARY KEY,
    Name VARCHAR(50) NOT NULL,
    Description VARCHAR(200)
);

-- 权限表
CREATE TABLE Permission (
    Id INT PRIMARY KEY,
    Code VARCHAR(100) NOT NULL UNIQUE,
    Name VARCHAR(50) NOT NULL,
    Module VARCHAR(50),
    Description VARCHAR(200)
);

-- 角色权限关联表
CREATE TABLE RolePermission (
    RoleId INT,
    PermissionId INT,
    PRIMARY KEY (RoleId, PermissionId)
);

-- 用户角色关联表
CREATE TABLE UserRole (
    UserId INT,
    RoleId INT,
    PRIMARY KEY (UserId, RoleId)
);
```

### 查询用户权限

```csharp
public class PermissionService : IPermissionService
{
    private readonly IUserContext _userContext;

    public async Task<bool> HasPermissionAsync(string permissionCode)
    {
        var userId = _userContext.UserId;
        return await HasPermissionAsync(userId, permissionCode);
    }

    public async Task<bool> HasPermissionAsync(int userId, string permissionCode)
    {
        // 查询用户是否拥有指定权限
        return await _dbContext.UserPermissions
            .Where(up => up.UserId == userId)
            .Where(up => up.Permission.Code == permissionCode)
            .AnyAsync();
    }

    public async Task<List<string>> GetUserPermissionsAsync(int userId)
    {
        return await _dbContext.UserPermissions
            .Where(up => up.UserId == userId)
            .Select(up => up.Permission.Code)
            .ToListAsync();
    }
}
```

## 前端权限控制

### 获取用户权限

```csharp
[HttpGet("permissions")]
[AppAuthorize("User.View")]
public async Task<IActionResult> GetMyPermissions()
{
    var userId = App.CurrentUser?.Id;
    var permissions = await _permissionService.GetUserPermissionsAsync(int.Parse(userId));
    return Ok(permissions);
}
```

### 前端存储和使用

```javascript
// 登录后保存权限
const permissions = response.data;
localStorage.setItem('permissions', JSON.stringify(permissions));

// 权限检查函数
function hasPermission(code) {
    const permissions = JSON.parse(localStorage.getItem('permissions') || '[]');
    return permissions.includes(code);
}

// 按钮级别控制
if (hasPermission('User.Delete')) {
    showDeleteButton();
}

// 路由级别控制
const protectedRoutes = [
    { path: '/users', permission: 'User.View' },
    { path: '/users/create', permission: 'User.Create' },
    { path: '/users/:id/edit', permission: 'User.Update' }
];
```

## 最佳实践

1. **权限编码统一规范**：使用 `模块.操作` 格式
2. **细粒度设计**：避免过粗的权限划分
3. **缓存权限数据**：避免频繁查询数据库
4. **前端双重验证**：前端显示控制 + 后端接口验证
5. **审计日志**：记录权限检查日志
6. **异常处理**：无权限时返回明确的错误信息
