# 快速开始

本指南将帮助你在最短的时间内上手 JBase.Common。

## 环境要求

- .NET Standard 2.1+
- ASP.NET Core 3.1+ / .NET 5+ / .NET 6+ / .NET 7+ / .NET 8+

## 安装

通过 NuGet 安装：

```bash
dotnet add package JBase.Common
```

## 标准项目结构

```
MyProject/
├── Program.cs                    # 程序入口
├── appsettings.json              # 配置文件
├── ServiceCollectionExtensions.cs # 服务注册扩展类
├── MiddlewareExtensions.cs       # 中间件扩展类
├── Controllers/
│   └── UserController.cs
└── Jobs/
    └── DataSyncJob.cs
```

## 完整示例

### 1. Program.cs

```csharp
using JBase.Common;
using JBase.Common.Extensions;
using JBase.Common.Middleware;

namespace MyApp.API
{
    public class Program
    {
        public static void Main(string[] args)
        {
            // 1. 创建Web应用构建器
            var builder = WebApplication.CreateBuilder(args);

            // 2. 初始化全局静态配置
            App.InitializeConfiguration(builder.Configuration);

            // 3. 注册项目所有服务
            builder.Services.AddProjectServices(builder.Environment);

            // 4. 构建应用实例
            var app = builder.Build();
            App.InitializeServices(app.Services);

            // 5. 配置中间件管道
            app.ConfigureMiddleware();

            // 6. 启动应用
            app.Run();
        }
    }
}
```

### 2. AppInitializer.cs（全局静态配置初始化）

```csharp
using JBase.Common;

namespace MyApp.API
{
    /// <summary>
    /// 全局静态配置初始化
    /// </summary>
    public static class AppInitializer
    {
        public static void InitializeStaticConfiguration(IConfiguration configuration)
        {
            App.InitializeConfiguration(configuration, new AppInitOptions
            {
                JwtOptionsKey = "Jwt",
                SwaggerConfigKey = "Swagger",
                CacheConfigKey = "Cache",
                AllowCorsKey = "AllowCors"
            });
        }
    }
}
```

### 3. ServiceCollectionExtensions.cs（服务注册）

```csharp
using JBase.Common;
using JBase.Common.Extensions;
using JBase.Common.Filters;
using JBase.Common.Job;
using JBase.Common.LogModule;
using JBase.Common.Middleware;
using Microsoft.Extensions.Logging;
using NLog.Extensions.Logging;

namespace MyApp.API
{
    /// <summary>
    /// 服务注册扩展类
    /// </summary>
    public static class ServiceCollectionExtensions
    {
        public static IServiceCollection AddProjectServices(
            this IServiceCollection services,
            IWebHostEnvironment environment)
        {
            #region 日志配置
            services.AddLogging(builder => builder.AddNLog());
            #endregion

            #region 控制器&全局过滤器
            services.AddControllers(options =>
            {
                // 全局验证特性
                options.Filters.Add<ModelValidationFilter>();

                // 请求参数处理过滤器
                options.Filters.Add<RequestFilter>();

                // 操作日志过滤器
                options.Filters.Add<OperationLogActionFilter>();
            })
            .AddJsonOptions(opt =>
            {
                opt.JsonSerializerOptions.Converters.Add(new DatetimeJsonConverter());
            });
            #endregion

            #region Swagger（开发环境）
            if (environment.IsDevelopment())
            {
                services.AddSwaggerGen();
            }
            #endregion

            #region 缓存服务
            services.AddCache();
            #endregion

            #region 定时任务
            services.AddJBaseJobs();
            JobAutoRegistrar.RegisterJobs();
            #endregion

            #region 全局异常中间件
            services.AddExceptionMiddleware<DatabaseExceptionHandler>(option =>
            {
                option.EnableCustomExceptionHandling = true;
            });
            #endregion

            #region 防止高频点击
            services.AddPreventRapidFireFilter();
            #endregion

            #region 防止重复提交
            services.AddPreventDuplicateRequestsFilter();
            #endregion

            #region 批量服务自动注册
            services.BatchRegisterService(x => x.FullName!.ToLower().Contains("myapp"));
            #endregion

            return services;
        }
    }
}
```

### 4. MiddlewareExtensions.cs（中间件配置）

```csharp
using JBase.Common.Middleware;

namespace MyApp.API
{
    /// <summary>
    /// 中间件管道扩展类
    /// </summary>
    public static class MiddlewareExtensions
    {
        public static WebApplication ConfigureMiddleware(this WebApplication app)
        {
            #region 基础中间件
            app.UseTraceId();
            app.UseCors();
            app.UseExceptionMiddleware();
            #endregion

            #region SwaggerUI
            if (app.Environment.IsDevelopment())
            {
                app.UseSwagger();
                app.UseSwaggerUI();
            }
            #endregion

            #region 路由
            app.UseRouting();
            #endregion

            #region API鉴权（仅对 /api 路径生效）
            app.UseWhen(ctx => ctx.Request.Path.StartsWithSegments("/api"), apiApp =>
            {
                apiApp.UseJwt();
                apiApp.UseMiddleware<CustomAuthorizeMiddleware>();
            });
            #endregion

            #region 终结点映射
            app.MapControllers();
            #endregion

            return app;
        }
    }
}
```

  }
