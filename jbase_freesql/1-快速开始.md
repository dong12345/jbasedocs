# 快速开始

## 安装 NuGet 包

```bash
dotnet add package JBase.FreeSql.Core
```

## 1. 定义实体

继承对应的实体基类：

```csharp
using JBase.FreeSql.Core.Entity;

// 方式一：使用带审计的实体（推荐）
public class User : AuditEntity
{
    public string Name { get; set; }
    public string Email { get; set; }
    public int Age { get; set; }
}

// 方式二：使用 int 主键的简单实体
public class Category : BaseEntity
{
    public string Name { get; set; }
    public string Description { get; set; }
}

// 方式三：使用 Guid 主键
public class Order : BaseGuidEntity
{
    public string OrderNo { get; set; }
    public decimal TotalAmount { get; set; }
}

// 方式四：自定义主键类型
public class Product : BaseEntity<long>
{
    public string Name { get; set; }
    public decimal Price { get; set; }
}
```

## 2. 配置服务

在 `Program.cs` 中配置：

```csharp
using JBase.FreeSql.Core.Extensions;
using JBase.FreeSql.Core.Options;

var builder = WebApplication.CreateBuilder(args);

// 添加 FreeSql 服务
builder.Services.AddJBaseFreeSql<int>(
    DataType.MySql,  // 数据库类型
    "Server=localhost;Database=mydb;User=root;Password=123456;",  // 连接字符串
    new AuditOptions
    {
        DefaultCreatedBy = "0",  // 默认创建人
        DefaultUpdatedBy = "0"   // 默认更新人
    },
    new EnumMappingOptions
    {
        DefaultStrategy = EnumMappingStrategy.Int  // 枚举映射为整数
    }
);

var app = builder.Build();

// 初始化 JBaseFreeSql
app.UseJBaseFreeSql();

app.Run();
```

## 3. 使用仓储

### 方式一：通过服务层使用

```csharp
public class UserService : BaseService<User>
{
    // 继承后自动拥有 CRUD 方法
}

public class MyService
{
    private readonly UserService _userService;

    public MyService(UserService userService)
    {
        _userService = userService;
    }

    public async Task Example()
    {
        // 插入
        var user = new User { Name = "张三", Email = "zhangsan@example.com", Age = 25 };
        var id = await _userService.InsertIdentityAsync(user);

        // 查询单个
        var u = await _userService.GetAsync(1);

        // 查询列表
        var list = await _userService.GetListAsync(x => x.Age > 20);

        // 分页查询
        var page = await _userService.GetPageAsync(1, 10, x => x.Age > 20);

        // 更新
        u.Age = 26;
        await _userService.UpdateAsync(u);

        // 删除（软删除）
        await _userService.DeleteAsync(1);
    }
}
```

### 方式二：直接使用仓储

```csharp
public class MyRepository
{
    private readonly IBaseRepository<User> _userRepo;

    public MyRepository(IBaseRepository<User> userRepo)
    {
        _userRepo = userRepo;
    }

    public async Task Example()
    {
        // 查询
        var user = await _userRepo.GetAsync(1);
        var users = await _userRepo.GetListAsync(x => x.Name.Contains("张"));

        // 插入
        await _userRepo.InsertAsync(new User { Name = "李四", Email = "lisi@example.com" });

        // 更新
        user.Age = 30;
        await _userRepo.UpdateAsync(user);

        // 删除
        await _userRepo.DeleteAsync(1);
    }
}
```

## 4. DTO 映射

```csharp
// 定义 DTO
public class UserDto
{
    public int Id { get; set; }
    public string Name { get; set; }
    public string Email { get; set; }
}

// 查询时直接转换为 DTO
var dto = await _userService.GetAsync<UserDto>(1);

// 查询列表时转换为 DTO
var dtos = await _userService.GetListAsync<UserDto>(x => x.Age > 20);
```

## 5. 事务处理

```csharp
// 使用扩展方法
var orm = app.Services.GetRequiredService<IFreeSql>();

await orm.TransactionAsync(async () =>
{
    await _userService.InsertAsync(new User { Name = "王五" });
    await _orderService.InsertAsync(new Order { OrderNo = "20240101" });
});

// 自定义事务隔离级别
await orm.TransactionAsync(async () =>
{
    // 操作...
}, IsolationLevel.ReadCommitted);
```

## 6. 复杂查询

```csharp
// 使用 AsSelect 获取 ISelect 进行复杂查询
var query = _userService.Repository.AsSelect()
    .Where(x => x.Age > 20)
    .WhereIf(true, x => x.Name.Contains("张"))
    .OrderByDescending(x => x.CreatedDate)
    .Select(x => new { x.Id, x.Name, x.Email });

var results = await query.ToListAsync();
```
